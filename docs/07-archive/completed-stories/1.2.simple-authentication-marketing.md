### Database Integration

```typescript
// Lead capture with Supabase
// src/lib/supabase/leads.ts
export async function captureLead(data: {
  email: string;
  source: string;
  utm_params?: UTMParams;
  gdpr_consent: boolean;
}) {
  const { data: lead, error } = await supabase
    .from('leads')
    .insert({
      email: data.email,
      source: data.source,
      utm_source: data.utm_params?.utm_source,
      utm_medium: data.utm_params?.utm_medium,
      utm_campaign: data.utm_params?.utm_campaign,
      gdpr_consent: data.gdpr_consent,
      gdpr_consent_date: new Date().toISOString(),
      gdpr_consent_ip: await getUserIP(),
    })
    .select()
    .single();

  return { lead, error };
}
```

### Auth Flow Architecture

```typescript
// Supabase Auth Setup
// src/lib/supabase/auth.ts

// Simple registration (demo access)
export async function registerForDemo(email: string, password: string) {
  // 1. Sign up user
  const { data: authData, error: authError } = await supabase.auth.signUp({
    email,
    password,
    options: {
      emailRedirectTo: `${process.env.NEXT_PUBLIC_APP_URL}/demo/welcome`,
    },
  });

  // 2. Create basic_users entry
  if (authData?.user) {
    await supabase.from('basic_users').insert({
      id: authData.user.id,
      email: authData.user.email,
      demo_access_granted_at: new Date().toISOString(),
      demo_access_expires_at: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000).toISOString(),
    });
  }

  return { authData, authError };
}
```
